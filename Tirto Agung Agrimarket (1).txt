            <div className="p-4 bg-slate-900 text-white flex items-center justify-between">
              <div className="flex items-center gap-2">
                <ShoppingBag className="w-5 h-5 text-emerald-400" />
                <h3 className="font-bold text-base">Keranjang Belanja</h3>
              </div>
              <button onClick={() => setIsCartOpen(false)} className="text-slate-400 hover:text-white">
                <X className="w-5 h-5" />
              </button>
            </div>

            <div className="flex-1 overflow-y-auto p-4 space-y-3">
              {cart.length === 0 ? (
                <div className="text-center py-16 text-slate-400 space-y-2">
                  <ShoppingCart className="w-12 h-12 mx-auto text-slate-300" />
                  <p className="text-xs font-bold">Keranjang Anda masih kosong.</p>
                </div>
              ) : (
                cart.map((item) => (
                  <div key={item.product.id} className="p-3 bg-slate-50 rounded-2xl border border-slate-200 flex items-center gap-3">
                    <SafeImage src={item.product.image} category={item.product.category} className="w-14 h-14 rounded-xl object-cover" />
                    <div className="flex-1 min-w-0">
                      <h5 className="font-bold text-xs text-slate-900 truncate">{item.product.name}</h5>
                      <p className="text-xs font-black text-emerald-700">Rp {item.product.price.toLocaleString('id-ID')} / {item.product.unit}</p>
                      
                      <div className="flex items-center gap-2 mt-2">
                        <button onClick={() => updateCartQty(item.product.id, -1)} className="w-6 h-6 rounded bg-slate-200 text-slate-700 font-bold text-xs flex items-center justify-center">-</button>
                        <span className="text-xs font-bold w-6 text-center">{item.qty}</span>
                        <button onClick={() => updateCartQty(item.product.id, 1)} className="w-6 h-6 rounded bg-slate-200 text-slate-700 font-bold text-xs flex items-center justify-center">+</button>
                      </div>
                    </div>
                  </div>
                ))
              )}
            </div>

            {cart.length > 0 && (
              <div className="p-4 border-t border-slate-200 bg-slate-50 space-y-3">
                <div className="flex justify-between text-xs text-slate-600">
                  <span>Subtotal Komoditas:</span>
                  <span className="font-bold text-slate-900">Rp {cartTotal.toLocaleString('id-ID')}</span>
                </div>
                <div className="flex justify-between text-xs text-slate-600">
                  <span>Ongkos Kirim Wilayah Kediri:</span>
                  <span className="font-bold text-slate-900">Rp 15.000</span>
                </div>
                <div className="flex justify-between text-sm font-black text-slate-900 pt-2 border-t border-slate-200">
                  <span>Total Tagihan:</span>
                  <span className="text-emerald-700">Rp {(cartTotal + 15000 + 2000).toLocaleString('id-ID')}</span>
                </div>

                <button
                  onClick={handleCheckout}
                  className="w-full py-3 bg-emerald-600 hover:bg-emerald-700 text-white font-extrabold text-xs rounded-xl shadow transition"
                >
                  Proses Pembuatan Pesanan
                </button>
              </div>
            )}

          </div>
        </div>
      )}

      {}
      {invoiceOrder && (
        <div className="fixed inset-0 z-50 bg-slate-950/70 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white rounded-3xl max-w-lg w-full p-6 space-y-4 border border-slate-200 shadow-2xl">
            
            <div className="flex items-center justify-between border-b border-slate-100 pb-3">
              <div className="flex items-center gap-2">
                <FileText className="w-6 h-6 text-emerald-600" />
                <h3 className="font-extrabold text-base text-slate-900">Invoice Resmi Tirto Agung</h3>
              </div>
              <button onClick={() => setInvoiceOrder(null)} className="text-slate-400 hover:text-slate-600">
                <X className="w-5 h-5" />
              </button>
            </div>

            <div className="space-y-2 text-xs">
              <div className="flex justify-between">
                <span className="text-slate-400">Nomor Pesanan:</span>
                <span className="font-bold text-slate-800">{invoiceOrder.id}</span>
              </div>
              <div className="flex justify-between">
                <span className="text-slate-400">Tanggal Transaksi:</span>
                <span className="font-bold text-slate-800">{invoiceOrder.date}</span>
              </div>
              <div className="flex justify-between">
                <span className="text-slate-400">Pembeli:</span>
                <span className="font-bold text-slate-800">{invoiceOrder.buyerName}</span>
              </div>
              <div className="flex justify-between">
                <span className="text-slate-400">Model Pembukuan:</span>
                <span className="font-bold text-sky-600">{invoiceOrder.model}</span>
              </div>
            </div>

            <div className="border border-slate-200 rounded-2xl p-3 space-y-2">
              <span className="text-[11px] font-bold text-slate-500">Rincian Barang:</span>
              {invoiceOrder.items.map((it, idx) => (
                <div key={idx} className="flex justify-between text-xs">
                  <span className="text-slate-800 font-medium">{it.name} x{it.qty}</span>
                  <span className="font-bold text-slate-900">Rp {(it.price * it.qty).toLocaleString('id-ID')}</span>
                </div>
              ))}
            </div>

            <div className="space-y-1 text-xs pt-2 border-t border-slate-100">
              <div className="flex justify-between text-slate-600">
                <span>Subtotal:</span>
                <span>Rp {invoiceOrder.subtotal.toLocaleString('id-ID')}</span>
              </div>
              <div className="flex justify-between text-slate-600">
                <span>Ongkir Kediri:</span>
                <span>Rp {invoiceOrder.shipping.toLocaleString('id-ID')}</span>
              </div>
              <div className="flex justify-between font-black text-sm text-slate-900 pt-1">
                <span>Total Lunas:</span>
                <span className="text-emerald-700">Rp {invoiceOrder.total.toLocaleString('id-ID')}</span>
              </div>
            </div>

            <button
              onClick={() => window.print()}
              className="w-full py-2.5 bg-slate-900 hover:bg-slate-800 text-white font-bold text-xs rounded-xl transition flex items-center justify-center gap-2"
            >
              <Printer className="w-4 h-4 text-emerald-400" /> Cetak Dokumen Invoice
            </button>

          </div>
        </div>
      )}

      {}
      {toast && (
        <div className="fixed bottom-20 md:bottom-6 right-6 z-50 bg-slate-900 text-white px-4 py-3 rounded-2xl shadow-xl flex items-center gap-3 border border-slate-800">
          <CheckCircle2 className="w-5 h-5 text-emerald-400 shrink-0" />
          <div>
            <h5 className="font-bold text-xs text-white">{toast.title}</h5>
            <p className="text-[11px] text-slate-300">{toast.message}</p>
          </div>
        </div>
      )}

      {}
      <div className="md:hidden fixed bottom-0 left-0 right-0 bg-slate-900 border-t border-slate-800 z-40 px-4 py-2 flex items-center justify-around text-slate-400">
        <button
          onClick={() => setActiveTab('katalog')}
          className={`flex flex-col items-center gap-1 text-[10px] font-bold ${
            activeTab === 'katalog' ? 'text-emerald-400' : ''
          }`}
        >
          <Store className="w-5 h-5" /> Katalog
        </button>
        <button
          onClick={() => setActiveTab('lapak')}
          className={`flex flex-col items-center gap-1 text-[10px] font-bold ${
            activeTab === 'lapak' ? 'text-emerald-400' : ''
          }`}
        >
          <Tag className="w-5 h-5" /> Lapak
        </button>
        <button
          onClick={() => setIsSellModalOpen(true)}
          className="flex flex-col items-center justify-center w-10 h-10 rounded-full bg-emerald-500 text-slate-950 font-bold shadow-lg"
        >
          <Plus className="w-6 h-6" />
        </button>
        <button
          onClick={() => setActiveTab('pesanan')}
          className={`flex flex-col items-center gap-1 text-[10px] font-bold ${
            activeTab === 'pesanan' ? 'text-emerald-400' : ''
          }`}
        >
          <Package className="w-5 h-5" /> Pesanan
        </button>
        <button
          onClick={() => setIsAuthModalOpen(true)}
          className="flex flex-col items-center gap-1 text-[10px] font-bold"
        >
          <User className="w-5 h-5" /> Akun
        </button>
      </div>

    </div>
  );
}
